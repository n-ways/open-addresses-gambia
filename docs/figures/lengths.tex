% Lengths in characters: (a) the plate names of the 2,347 places, (b) the home forms of every issued code.
% Data: lengths-plate.dat, lengths-home.dat.
\documentclass[border=2mm]{standalone}
\usepackage{fontspec}\setmainfont{LibertinusSans}[Extension=.otf,UprightFont=*-Regular,BoldFont=*-Bold]
\usepackage{pgfplots}\pgfplotsset{compat=1.18, /pgf/number format/assume math mode=true}
\definecolor{ink}{HTML}{0C1C8C}\definecolor{flagred}{HTML}{CE1126}\definecolor{flaggreen}{HTML}{3A7728}
\usepgfplotslibrary{groupplots}
% \tr{English}{French}: the figure's language; <name>-fr.tex sets \figfr and inputs this file.
\ifdefined\figfr\newcommand{\tr}[2]{#2}\pgfkeys{/pgf/number format/.cd, use comma, 1000 sep={\,}}\else\newcommand{\tr}[2]{#1}\fi
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[group style={group size=2 by 1, horizontal sep=1.6cm}, width=7.6cm, height=5.4cm, ybar, ymin=0,
  tick label style={font=\scriptsize}, label style={font=\small}, title style={font=\small}, axis lines*=left,
  ymajorgrids, grid style={black!12}, scaled y ticks=false]
\nextgroupplot[title={\tr{(a) plate names}{(a) noms de plaque}}, xlabel={\tr{characters}{caractères}}, ylabel={\tr{places}{lieux}}, bar width=0.28cm, xtick={2,4,...,12}, xmin=1.3, xmax=12.7]
\addplot[fill=ink, draw=none] table[x=chars, y=places] {lengths-plate.dat};
\draw[flagred, dashed] (axis cs:12.5,0) -- (axis cs:12.5,430) node[left, font=\scriptsize, pos=0.97] {\tr{limit}{limite}};
\nextgroupplot[title={\tr{(b) home forms, e.g.\ \emph{Bakau 2389}}{(b) formes domicile, p.\,ex.\ \emph{Bakau 2389}}}, xlabel={\tr{characters, space included}{caractères, espaces comprises}}, ylabel={\tr{codes (thousands)}{codes (milliers)}},
  bar width=0.25cm, xtick={7,9,...,19}, xmin=6.3, xmax=19.7, y filter/.expression={y/1000}]
\addplot[fill=flaggreen, draw=none] table[x=chars, y=codes] {lengths-home.dat};
\end{groupplot}
\end{tikzpicture}
\end{document}
